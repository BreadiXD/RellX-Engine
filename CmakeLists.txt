cmake_minimum_required(VERSION 3.22)
project(RellXEngine)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Librarys
# SDL3
set(SDL3_PATH "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/SDL3-3.4.4")
set(SDL_SHARED OFF CACHE BOOL "Dont Build Shared SDL3" FORCE)
set(SDL_STATIC ON CACHE BOOL "Build Static SDL3" FORCE)

set(SDL_GPU OFF CACHE BOOL "Stop SDL_GPU" FORCE)
set(SDL_RENDER_GPU OFF CACHE BOOL "Stop SDL_RENDER_GPU" FORCE)
set(SDL_RENDER OFF CACHE BOOL "Stop SDL_RENDER" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "Stop SDL_CAMERA" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "Stop SDL_SENSOR" FORCE)

add_subdirectory(${SDL3_PATH})
# GLM
set(GLM_PATH "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/GLM-1.0.3")
set(GLM_ENABLE_CXX_17 ON CACHE BOOL "Enable GLM GLM_ENABLE_CXX_17" FORCE)
set(GLM_ENABLE_SIMD_SSE3 ON CACHE BOOL "Enable GLM GLM_ENABLE_SIMD_SSE3" FORCE)

add_subdirectory(${GLM_PATH})
# Rpmalloc
set(RPMALLOC_PATH "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/RPMALLOC-1.4.5")

add_subdirectory(${RPMALLOC_PATH})





# Output Path
set(BIN_PATH "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(NOT EXISTS "${BIN_PATH}")
    file(MAKE_DIRECTORY "${BIN_PATH}")
endif()
set(EXECUTABLE_OUTPUT_PATH "${BIN_PATH}")


file(GLOB_RECURSE SOURCES_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Src/*.cpp")
add_executable(${PROJECT_NAME} main.cpp ${SOURCES_FILES})
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra)
if (CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_definitions(${PROJECT_NAME} PRIVATE 
        RELLX_DEBUG_LOGGING
    )
elseif(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_definitions(${PROJECT_NAME} PRIVATE NDEBUG)
    target_compile_options(${PROJECT_NAME} PRIVATE -O3)
endif()

target_precompile_headers(${PROJECT_NAME} PRIVATE "Include/RellXLib.hpp")
target_link_options(${PROJECT_NAME} PRIVATE -static -static-libgcc)

target_link_libraries(${PROJECT_NAME} PRIVATE 
    SDL3-static
    glm
    rpmalloc
)
target_include_directories(${PROJECT_NAME} PRIVATE 
    "${RPMALLOC_PATH}/rpmalloc"
    "${GLM_PATH}/GLM-1.0.3"
    "${SDL3_PATH}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/Include"
)
